\[ \mathrm{CircuitBasedQuantumComputing.Exact}\ \mathrm{QuantumGate.CNOT}\ \mathrm{CircuitBasedQuantumComputing.ruleCNOT}\ 1 \]

\paragraph{\texttt{CircuitBasedQuantumComputing.stabilizer\_hadamard}}
