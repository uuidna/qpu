\paragraph{\texttt{paulis\_growth}}
One more qubit quadruples the phased Pauli group, for every nn.
\[ \forall (\mathrm{nn} : \mathbb{N}), \mathrm{paulis} (\mathrm{nn} + 1) = 4 \cdot \mathrm{paulis}\ \mathrm{nn} \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Stabilizer.lean\#L51}

\paragraph{\texttt{paulis\_pos}}
The phased Pauli group is nonempty, for every nn.
\[ \forall (\mathrm{nn} : \mathbb{N}), 0 < \mathrm{paulis}\ \mathrm{nn} \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Stabilizer.lean\#L56}

\paragraph{\texttt{stabProd\_grows}}
The stabilizer product never shrinks across a factor, for every k.
\[ \forall (k : \mathbb{N}), \mathrm{stabProd}\ k \le \mathrm{stabProd} (k + 1) \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Stabilizer.lean\#L42}

\paragraph{\texttt{stabProd\_pos}}
The stabilizer product never vanishes, for every k.
\[ \forall (k : \mathbb{N}), 0 < \mathrm{stabProd}\ k \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Stabilizer.lean\#L34}

\paragraph{\texttt{stabProd\_step}}
One stabilizer factor, for every k.
\[ \forall (k : \mathbb{N}), \mathrm{stabProd} (k + 1) = (\mathrm{mint} (k + 1) + 1) \cdot \mathrm{stabProd}\ k \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Stabilizer.lean\#L31}

\paragraph{\texttt{stabStates\_pos}}
n qubits always have stabilizer states, for every n.
\[ \forall (\mathrm{nn} : \mathbb{N}), 0 < \mathrm{stabStates}\ \mathrm{nn} \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Stabilizer.lean\#L47}

\paragraph{\texttt{stabSubgroup\_doubles}}
One more qubit doubles the stabilizer subgroup, for every nn.
\[ \forall (\mathrm{nn} : \mathbb{N}), \mathrm{stabSubgroup} (\mathrm{nn} + 1) = 2 \cdot \mathrm{stabSubgroup}\ \mathrm{nn} \]
