\paragraph{\texttt{fusionDim\_doubles}}
One more fusion level exactly doubles the dimension, for every nn.
\[ \forall (\mathrm{nn} : \mathbb{N}), \mathrm{fusionDim} (\mathrm{nn} + 1) = 2 \cdot \mathrm{fusionDim}\ \mathrm{nn} \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Fusion.lean\#L35}

\paragraph{\texttt{fusionDim\_pos}}
Fusion dimension is positive, for every nn.
\[ \forall (\mathrm{nn} : \mathbb{N}), 0 < \mathrm{fusionDim}\ \mathrm{nn} \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Fusion.lean\#L24}

\paragraph{\texttt{fusionEntropy\_doubles}}
One more fusion level exactly doubles the entropy, for every nn.
\[ \forall (\mathrm{nn} : \mathbb{N}), \mathrm{fusionEntropy} (\mathrm{nn} + 1) = 2 \cdot \mathrm{fusionEntropy}\ \mathrm{nn} \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Fusion.lean\#L40}

\paragraph{\texttt{fusionEntropy\_pos}}
Fusion entropy is positive, for every nn.
\[ \forall (\mathrm{nn} : \mathbb{N}), 0 < \mathrm{fusionEntropy}\ \mathrm{nn} \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Fusion.lean\#L27}

\paragraph{\texttt{fusionStates\_doubles}}
One more fusion level exactly doubles the state count, for every nn.
\[ \forall (\mathrm{nn} : \mathbb{N}), \mathrm{fusionStates} (\mathrm{nn} + 1) = 2 \cdot \mathrm{fusionStates}\ \mathrm{nn} \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Fusion.lean\#L30}

\paragraph{\texttt{fusionStates\_pos}}
Fusion states are positive, for every nn.
\[ \forall (\mathrm{nn} : \mathbb{N}), 0 < \mathrm{fusionStates}\ \mathrm{nn} \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/Processing/Fusion.lean\#L21}
