\paragraph{\texttt{Qpu.Federated.avgV\_bounded}}
Coordinate-wise bounded updates average to a coordinate-wise bounded vector.
\[ \forall \{d : \mathbb{N}\} (\mathrm{lo}\ \mathrm{hi} : \mathbb{Z}) (\mathrm{us} : \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d)), 0 < \mathrm{us.length} \to \forall (i : \mathrm{Fin}\ d), (\forall u \in \mathrm{us}, \mathrm{lo} \le u\ i \land u\ i \le \mathrm{hi}) \to \mathrm{lo} \le \mathrm{Qpu.Federated.avgV}\ \mathrm{us}\ i \land \mathrm{Qpu.Federated.avgV}\ \mathrm{us}\ i \le \mathrm{hi} \]
\noindent axioms: \texttt{propext}, \texttt{Classical.choice}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/MachineLearning/FederatedLearning.lean\#L151}

\paragraph{\texttt{Qpu.Federated.avgV\_replicate}}
Unbiasedness: averaging `k > 0` identical updates returns that update.
\[ \forall \{d : \mathbb{N}\} (k : \mathbb{N}), 0 < k \to \forall (u : \mathrm{Qpu.Federated.Vec}\ d), \mathrm{Qpu.Federated.avgV} (\mathrm{List.replicate}\ k\ u) = u \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/MachineLearning/FederatedLearning.lean\#L145}

\paragraph{\texttt{Qpu.Federated.avgZ\_bounded}}

\paragraph{\texttt{Qpu.Federated.shares\_consistent}}
Hiding one mask: whatever the observed head share and the other masks, every alternative secret `s'` admits a mask `m'` reproducing exactly those observed shares. Together with `shares\_tail\_independent` this is: any `k − 1` of `k` shares are consistent with every secret.
\[ \forall \{d : \mathbb{N}\} (s\ s'\ m : \mathrm{Qpu.Federated.Vec}\ d) (\mathrm{ms} : \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d)), \exists m', (\mathrm{Qpu.Federated.shares}\ s' (m' :: \mathrm{ms}))\mathrm{.head}? = (\mathrm{Qpu.Federated.shares}\ s (m :: \mathrm{ms}))\mathrm{.head}? \land (\mathrm{Qpu.Federated.shares}\ s' (m' :: \mathrm{ms}))\mathrm{.tail.tail} = (\mathrm{Qpu.Federated.shares}\ s (m :: \mathrm{ms}))\mathrm{.tail.tail} \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/MachineLearning/FederatedLearning.lean\#L251}

\paragraph{\texttt{Qpu.Federated.shares\_length}}
\[ \forall \{d : \mathbb{N}\} (s : \mathrm{Qpu.Federated.Vec}\ d) (\mathrm{ms} : \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d)), (\mathrm{Qpu.Federated.shares}\ s\ \mathrm{ms})\mathrm{.length} = \mathrm{ms.length} + 1 \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/MachineLearning/FederatedLearning.lean\#L236}

\paragraph{\texttt{Qpu.Federated.shares\_tail\_independent}}
The masks are chosen independently of the secret: the tail of the sharing never depends on `s`.
\[ \forall \{d : \mathbb{N}\} (s\ s' : \mathrm{Qpu.Federated.Vec}\ d) (\mathrm{ms} : \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d)), (\mathrm{Qpu.Federated.shares}\ s\ \mathrm{ms})\mathrm{.tail} = (\mathrm{Qpu.Federated.shares}\ s'\ \mathrm{ms})\mathrm{.tail} \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/MachineLearning/FederatedLearning.lean\#L245}

\paragraph{\texttt{Qpu.Federated.shares\_traffic}}
Traffic: with `k` shares per client, one round carries `clients · k` share messages.
\[ \forall \{d : \mathbb{N}\} (k : \mathbb{N}) (\mathrm{cs} : \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d \times \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d))), (\forall c \in \mathrm{cs}, \mathrm{c.2.length} + 1 = k) \to (\mathrm{List.flatMap} (\lambda \ c \mapsto \mathrm{Qpu.Federated.shares}\ \mathrm{c.1}\ \mathrm{c.2}) \mathrm{cs})\mathrm{.length} = \mathrm{cs.length} \cdot k \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/MachineLearning/FederatedLearning.lean\#L279}

\paragraph{\texttt{Qpu.Federated.sortZ\_length}}
\[ \forall (l : \mathrm{List} \mathbb{Z}), (\mathrm{Qpu.Federated.sortZ}\ l)\mathrm{.length} = \mathrm{l.length} \]

\paragraph{\texttt{Qpu.Federated.sortZ\_sorted}}

\paragraph{\texttt{Qpu.Federated.sorted\_lower}}
In a sorted list, if at most `k` values are below `lo`, the `k`-th value is at least `lo`.
\[ \forall \{s : \mathrm{List} \mathbb{Z}\}, \mathrm{List.Pairwise} (\lambda \ \mathrm{x1}\ \mathrm{x2} \mapsto \mathrm{x1} \le \mathrm{x2}) s \to \forall (\mathrm{lo} : \mathbb{Z}) \{k : \mathbb{N}\} (\mathrm{hk} : k < \mathrm{s.length}), \mathrm{List.countP} (\lambda \ x \mapsto \mathrm{decide} (x < \mathrm{lo})) s \le k \to \mathrm{lo} \le s[k] \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/MachineLearning/FederatedLearning.lean\#L315}

\paragraph{\texttt{Qpu.Federated.sorted\_upper}}
In a sorted list of length `n`, if at most `n − 1 − k` values exceed `hi`, the `k`-th value is at most `hi`.
\[ \forall \{s : \mathrm{List} \mathbb{Z}\}, \mathrm{List.Pairwise} (\lambda \ \mathrm{x1}\ \mathrm{x2} \mapsto \mathrm{x1} \le \mathrm{x2}) s \to \forall (\mathrm{hi} : \mathbb{Z}) \{k : \mathbb{N}\} (\mathrm{hk} : k < \mathrm{s.length}), \mathrm{List.countP} (\lambda \ x \mapsto \mathrm{decide} (\mathrm{hi} < x)) s \le \mathrm{s.length} - 1 - k \to s[k] \le \mathrm{hi} \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/MachineLearning/FederatedLearning.lean\#L338}

\paragraph{\texttt{Qpu.Federated.sumV\_append}}
Aggregation is linear over concatenation: the server may aggregate in batches.
\[ \forall \{d : \mathbb{N}\} (\mathrm{us}\ \mathrm{vs} : \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d)), \mathrm{Qpu.Federated.sumV} (\mathrm{us} ++ \mathrm{vs}) = \mathrm{Qpu.Federated.addV} (\mathrm{Qpu.Federated.sumV}\ \mathrm{us}) (\mathrm{Qpu.Federated.sumV}\ \mathrm{vs}) \]
\noindent axioms: \texttt{propext}, \texttt{Quot.sound}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/MachineLearning/FederatedLearning.lean\#L134}

\paragraph{\texttt{Qpu.Federated.sumV\_cons}}
\[ \forall \{d : \mathbb{N}\} (u : \mathrm{Qpu.Federated.Vec}\ d) (\mathrm{us} : \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d)), \mathrm{Qpu.Federated.sumV} (u :: \mathrm{us}) = \mathrm{Qpu.Federated.addV}\ u (\mathrm{Qpu.Federated.sumV}\ \mathrm{us}) \]

\paragraph{\texttt{Qpu.Federated.sumV\_perm}}
Aggregation is order-independent: any arrival order of the clients gives the same sum.
\[ \forall \{d : \mathbb{N}\} \{\mathrm{us}\ \mathrm{vs} : \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d)\}, \mathrm{us.Perm}\ \mathrm{vs} \to \mathrm{Qpu.Federated.sumV}\ \mathrm{us} = \mathrm{Qpu.Federated.sumV}\ \mathrm{vs} \]

\paragraph{\texttt{Qpu.Federated.sumV\_shares}}
The shares sum back to the secret.
\[ \forall \{d : \mathbb{N}\} (s : \mathrm{Qpu.Federated.Vec}\ d) (\mathrm{ms} : \mathrm{List} (\mathrm{Qpu.Federated.Vec}\ d)), \mathrm{Qpu.Federated.sumV} (\mathrm{Qpu.Federated.shares}\ s\ \mathrm{ms}) = s \]

\paragraph{\texttt{Qpu.Federated.sumZ\_append}}

\paragraph{\texttt{Qpu.Federated.sumZ\_bounded}}
Every element in `[lo, hi]` → the sum lies in `[lo·|l|, hi·|l|]`.
\[ \forall (\mathrm{lo}\ \mathrm{hi} : \mathbb{Z}) (l : \mathrm{List} \mathbb{Z}), (\forall x \in l, \mathrm{lo} \le x \land x \le \mathrm{hi}) \to \mathrm{lo} \cdot \uparrow \mathrm{l.length} \le \mathrm{Qpu.Federated.sumZ}\ l \land \mathrm{Qpu.Federated.sumZ}\ l \le \mathrm{hi} \cdot \uparrow \mathrm{l.length} \]
