\paragraph{\texttt{Supremacy.lanes\_le\_cores}}
\[ \forall \{\mathrm{cores} : \mathbb{N}\} (\mathrm{share} : \mathbb{N}), 1 \le \mathrm{cores} \to \mathrm{max}\ 1 (\mathrm{min}\ \mathrm{cores}\ \mathrm{share}) \le \mathrm{cores} \]
\noindent axioms: \texttt{propext}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/Supremacy.lean\#L35}

\paragraph{\texttt{Supremacy.lanes\_pos}}
\[ \forall (\mathrm{share} : \mathbb{N}), 1 \le \mathrm{max}\ 1\ \mathrm{share} \]
\noindent axioms: \texttt{propext}; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/Supremacy.lean\#L33}

\paragraph{\texttt{Supremacy.separation}}
\[ \forall \{\mathrm{tq}\ b\ \mathrm{tc} : \mathbb{N}\}, \mathrm{tq} \le b \to b < \mathrm{tc} \to \mathrm{tq} < \mathrm{tc} \]
\noindent axioms: none; proof: \url{https://github.com/uuidna/qpu/blob/v1.1.1/lean/Qpu/General/Supremacy.lean\#L25}
